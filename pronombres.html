θi = tu inform. | θu | θin | θes | θek = contigo

ci = el | cu | cin | ces | cik = con el
ŝi = ella | ŝu | ŝin | ŝes | ŝik = con ella
li = alguien / el-ella | lu | lin | les | lik = con el
oli = alguien | olu | olin | oles | olik = con alguien
eli = este / estos | elu | elin | eles | —

  colectivo adicional visible: olĉi = cual de los dos?

ARTÍCULOS Y SERIE -o / -i

to = el inanimado | te | ton | tes | —
  endo = este (de otros)
  oldo = el otro

ti = los, inanimado | tos | tof | tor | —

ĉom = cuanto?
  otom = tanto
  oĉom = todo
  plom = mucho
  jom = poco

ĉam = cuando? | — | ĉamin | ĉamel | —
  sam = ya, ahora
  tam = cuando...
  etam = entonces
  olam = aveces
  nevam = nunca
  pojam = tarde, despues

ĉok = donde? | — | ĉojn | ĉojel | —
  sok = aqui
  tok = donde..
  elok = alla
  somok = en algun lugar

ĉor = de que forma? como?
  oĉor = de tantas formas
  sor = asi
  nelor = de ninguna manera

  taj = porque...
