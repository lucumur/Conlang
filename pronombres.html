Transcripción de los paradigmas documentados. Las casillas ausentes no se completan por inferencia.

PRONOMBRES -i

mi = yo | ma / mu | min | mes | mek = conmigo
vi = tu | va / vu | vin | ves | vek = contigo
θi = tu inform. | θu | θin | θes | θek = contigo
ži = vos (fem) | žu | žin | žes | vožek = con vos
ci = el | cu | cin | ces | cik = con el
ši = ella | šu | šin | šes | šik = con ella
li = alguien / el-ella | lu | lin | les | lik = con el
oli = alguien | olu | olin | oles | olik = con alguien
eli = este / estos | elu | elin | eles | —
či = que?, el que | ču | čin | čes | ček = con eso
  colectivo adicional visible: olči = cual de los dos?
ečki = cada uno, cada... | ečku | — | — | —
ĝi = ese animado | ĝu | ĝin | ĝes | —
ri = algo, anything, lo que sea | ru | rin | res | rem = algo
ori = algo | oru | orin | ores | —
neri = ninguno | neru | nerin | neres | nem = nada

vori = vuestro | vora | voron | voras | —

ARTÍCULOS Y SERIE -o / -i

to = el inanimado | te | ton | tes | —
  endo = este (de otros)
  oldo = el otro

ti = los, inanimado | tos | tof | tor | —

čom = cuanto?
  otom = tanto
  očom = todo
  plom = mucho
  jom = poco

čam = cuando? | — | čamin | čamel | —
  sam = ya, ahora
  tam = cuando...
  etam = entonces
  olam = aveces
  nevam = nunca
  pojam = tarde, despues

čok = donde? | — | čojn | čojel | —
  sok = aqui
  tok = donde..
  elok = alla
  somok = en algun lugar

čor = de que forma? como?
  očor = de tantas formas
  sor = asi
  nelor = de ninguna manera

čaj = porque?
  čajpor = para que
  taj = porque...
